Database Conception (MongoDB style)
Users

Users {
  _id: ObjectId,
  name: String,
  email: String (unique),
  password: String,
  role: ["user", "coach", "admin"],

  // Athlete details
  age: Number,
  gender: String,
  height: Number,
  weight: Number,
  bmi: Number,
  fitnessLevel: ["beginner", "intermediate", "advanced"],
  medicalConditions: [String],
  allergies: [String],
  activityFrequency: ["sedentary", "moderate", "active"],

  profile: {
    avatar: String,
    bio: String
  },
  createdAt: Date,
  updatedAt: Date
}
goals {
      types: [
      "weight_loss", 
      "muscle_gain", 
      "endurance",
       "general_fitness"
       ]
}

Programs
Programs {
  _id: ObjectId,
  title: String,
  type: ["muscle_gain", "weight_loss", "endurance"],
  description: String,
  coachId: ObjectId (ref: Users),
  duration: String,
  level: ["beginner", "intermediate", "advanced"],
  createdAt: Date,
  updatedAt: Date
}

ToDo (Weekly Planner)
ToDo {
  _id: ObjectId,
  userId: ObjectId (ref: Users),
  week: String,
  trainingPlan: [{ day: String, activity: String }],
  mealPlan: [{ day: String, meal: String }]
}

Products
Products {
  _id: ObjectId,
  name: String,
  category: ["supplement", "gear", "clothing"],
  price: Number,
  stock: Number,
  description: String,
  imageUrl: String,
  createdAt: Date,
  updatedAt: Date
}

Orders
Orders {
  _id: ObjectId,
  userId: ObjectId (ref: Users),
  products: [
    { productId: ObjectId (ref: Products), quantity: Number }
  ],
  total: Number,
  status: ["pending", "paid", "shipped"],
  createdAt: Date,
  updatedAt: Date
}

Payments
Payments {
  _id: ObjectId,
  userId: ObjectId (ref: Users),
  packType: ["monthly", "yearly"],
  amount: Number,
  status: ["pending", "paid"],
  createdAt: Date,
  updatedAt: Date
}

Equipments
Equipments {
  _id: ObjectId,
  name: String,
  category: ["cardio", "strength", "flexibility"],
  status: ["available", "under_maintenance", "out_of_order"],
  purchaseDate: Date,
  lastMaintenance: Date,
  location: String,
  imageUrl: String,
  createdAt: Date,
  updatedAt: Date
}

Repairs
Repairs {
  _id: ObjectId,
  equipmentId: ObjectId (ref: Equipments),
  reportedBy: ObjectId (ref: Users),
  issueDescription: String,
  status: ["pending", "in_progress", "fixed"],
  reportedAt: Date,
  fixedAt: Date,
  repairedBy: String
}

Comments
Comments {
  _id: ObjectId,
  userId: ObjectId (ref: Users),
  targetType: ["program", "coach", "product", "equipment", "event", "competition"],
  targetId: ObjectId,
  text: String,
  rating: Number (1-5),
  createdAt: Date,
  updatedAt: Date 
}

Events
Events {
  _id: ObjectId,
  title: String,
  description: String,
  date: Date,
  location: String,
  organizer: ObjectId (ref: Users),
  participants: [ObjectId (ref: Users)],
  category: ["workshop", "seminar", "marathon", "fitness_class"],
  createdAt: Date,
  updatedAt: Date
}

Competitions
Competitions {
  _id: ObjectId,
  title: String,
  description: String,
  date: Date,
  location: String,
  organizer: ObjectId (ref: Users),
  participants: [
    {
      userId: ObjectId (ref: Users),
      score: Number,
      rank: Number
    }
  ],
  category: ["bodybuilding", "cycling", "crossfit", "powerlifting"],
  prizes: [String],
  createdAt: Date,
  updatedAt: Date
}
